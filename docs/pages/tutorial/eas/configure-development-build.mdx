import {
  StyleSheet,
  Text,
  View,
  TextInput,
  TouchableOpacity,
  ScrollView,
  Alert
} from 'react-native';

  // DEĞİŞKENLER
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [rememberMe, setRememberMe] = useState(true);
  const [status, setStatus] = useState('🟢 Çevrimiçi');
  const [currentScreen, setCurrentScreen] = useState('login');
  const [messageText, setMessageText] = useState('');
  const [selectedContact, setSelectedContact] = useState('Ahmet Yılmaz');

  // GİRİŞ YAP

      return;
    }
    
    Alert.alert(
      'Hoş Geldiniz!',
      `MSN Messenger'a hoş geldiniz ${email.split('@')[0]}!`,
      [{ text: 'Tamam', onPress: () => setCurrentScreen('contacts') }]

  };

  // ÇIKIŞ YAP
  const handleLogout = () => {
    setEmail('');
    setPassword('');
    setCurrentScreen('login');
    Alert.alert('Başarılı', 'Çıkış yapıldı.');

    { id: 1, name: 'Ahmet Yılmaz', status: '🟢 Çevrimiçi', lastSeen: 'Şu anda' },
    { id: 2, name: 'Ayşe Demir', status: '🌙 Uzakta', lastSeen: '10 dk önce' },
    { id: 3, name: 'Mehmet Kaya', status: '🔴 Meşgul', lastSeen: 'Rahatsız etmeyin' },
    { id: 4, name: 'Fatma Şahin', status: '⚫ Çevrimdışı', lastSeen: '2 saat önce' },
  ];

  // DEMO MESAJLAR
  const demoMessages = [
    { id: 1, sender: 'Ahmet', text: 'Merhaba! Nasılsın?', time: '14:30', isMe: false },
    { id: 2, sender: 'Ben', text: 'İyiyim, teşekkürler! 😊', time: '14:31', isMe: true },
    { id: 3, sender: 'Ahmet', text: 'MSN geri geldi diye çok sevindim!', time: '14:32', isMe: false },
  ];

  // DURUM SEÇENEKLERİ
  const statusOptions = [
    '🟢 Çevrimiçi',
    '🌙 Uzakta', 
    '🔴 Meşgul',
    '📞 Telefondayım',
    '⚫ Çevrimdışı'
  ];

  // ---------- GİRİŞ EKRANI ----------
  if (currentScreen === 'login') {
    return (
      <ScrollView style={styles.container}>
        {/* MSN BAŞLIK */}
        <View style={styles.msnHeader}>
          <Text style={styles.msnLogo}>MSN</Text>
          <Text style={styles.messengerText}>Messenger</Text>
          <Text style={styles.subtitle}>2000'lerin Ruhuyla</Text>
        </View>

        {/* GİRİŞ FORMU */}
        <View style={styles.loginBox}>
          <Text style={styles.loginTitle}>📨 Oturum Aç</Text>
          
          <TextInput
            style={styles.input}
            placeholder="ornek@hotmail.com"
            placeholderTextColor="#888"
            value={email}
            onChangeText={setEmail}
          />
          
          <TextInput
            style={styles.input}
            placeholder="Şifre"
            placeholderTextColor="#888"
            value={password}
            onChangeText={setPassword}
            secureTextEntry
          />

          {/* DURUM SEÇ */}
          <Text style={styles.label}>Durumunuz:</Text>
          <View style={styles.statusContainer}>
            {statusOptions.map((option, index) => (
              <TouchableOpacity
                key={index}
                style={[
                  styles.statusButton,
                  status === option && styles.statusButtonActive
                ]}
                onPress={() => setStatus(option)}
              >
                <Text style={[
                  styles.statusButtonText,
                  status === option && { color: 'white' }
                ]}>
                  {option}
                </Text>
              </TouchableOpacity>
            ))}
          </View>

          {/* BENİ HATIRLA */}
          <TouchableOpacity
            style={styles.checkboxContainer}
            onPress={() => setRememberMe(!rememberMe)}
          >
            <View style={[styles.checkbox, rememberMe && styles.checkboxChecked]}>
              {rememberMe && <Text style={styles.checkmark}>✓</Text>}
            </View>
            <Text style={styles.checkboxLabel}>Beni hatırla</Text>
          </TouchableOpacity>

          {/* GİRİŞ BUTONU */}
          <TouchableOpacity style={styles.loginButton} onPress={handleLogin}>
            <Text style={styles.loginButtonText}>OTURUM AÇ</Text>
          </TouchableOpacity>

          <Text style={styles.note}>
            💡 Demo: Herhangi bir e-posta ile giriş yapabilirsiniz.
          </Text>
        </View>

        {/* ÖZELLİKLER */}
        <View style={styles.features}>
          <Text style={styles.featuresTitle}>🎮 MSN Özellikleri:</Text>
          <TouchableOpacity onPress={() => Alert.alert('Mesajlaşma', 'Anında mesajlaşma özelliği aktif!')}>
            <Text style={styles.feature}>• Anında Mesajlaşma</Text>
          </TouchableOpacity>
          <TouchableOpacity onPress={() => Alert.alert('Emoticons', 'Emoticon galerisi yakında!')}>
            <Text style={styles.feature}>• Emoticon Galerisi</Text>
          </TouchableOpacity>
          <TouchableOpacity onPress={() => Alert.alert('Dosya', 'Dosya transferi özelliği yakında!')}>
            <Text style={styles.feature}>• Dosya Transferi</Text>
          </TouchableOpacity>
          <TouchableOpacity onPress={() => Alert.alert('Durum', 'Kişisel durumlarınızı değiştirebilirsiniz!')}>
            <Text style={styles.feature}>• Kişisel Durumlar</Text>
          </TouchableOpacity>
          <TouchableOpacity onPress={() => Alert.alert('Nudge', 'Nudge özelliği aktif!')}>
            <Text style={styles.feature}>• Nudge (Sarsma)</Text>
          </TouchableOpacity>
        </View>

        <View style={styles.footer}>
          <Text style={styles.footerText}>© 2024 MSN Revival</Text>
        </View>
      </ScrollView>
    );

    return (
      <ScrollView style={styles.container}>
        {/* ÜST MENÜ */}
        <View style={styles.topBar}>
          <Text style={styles.topTitle}>MSN Messenger</Text>
          <View style={styles.topButtons}>
            <TouchableOpacity 
              style={styles.topButton}
              onPress={() => {
                const currentIndex = statusOptions.indexOf(status);
                const nextIndex = (currentIndex + 1) % statusOptions.length;
                setStatus(statusOptions[nextIndex]);
              }}
            >
              <Text style={styles.topButtonText}>{status}</Text>
            </TouchableOpacity>
            <TouchableOpacity style={styles.topButton} onPress={handleLogout}>
              <Text style={styles.topButtonText}>Çıkış</Text>
            </TouchableOpacity>
          </View>
        </View>

        {/* ANA MENÜ */}
        <View style={styles.mainMenu}>
          <TouchableOpacity 
            style={[styles.menuItem, currentScreen === 'contacts' && styles.activeMenuItem]}
            onPress={() => setCurrentScreen('contacts')}
          >
            <Text style={styles.menuText}>👥 Kişiler</Text>
          </TouchableOpacity>
          <TouchableOpacity 
            style={[styles.menuItem, currentScreen === 'chat' && styles.activeMenuItem]}
            onPress={() => {
              setSelectedContact('Ahmet Yılmaz');
              setCurrentScreen('chat');
            }}
          >
            <Text style={styles.menuText}>💬 Sohbet</Text>

            style={styles.menuItem}
            onPress={() => Alert.alert('Emoticons', 'Emoticon galerisi yakında! 😊')}

            <Text style={styles.menuText}>😊 Emoticons</Text>

            style={styles.menuItem}
            onPress={() => Alert.alert('Ayarlar', 'Ayarlar menüsü yakında! ⚙️')}
          >
            <Text style={styles.menuText}>⚙️ Ayarlar</Text>

              style={styles.contactItem}

              </View>

                  setCurrentScreen('chat');

            <Text style={styles.actionIcon}>➕</Text>
            <Text style={styles.actionText}>Kişi Ekle</Text>

            style={styles.actionButton}
            onPress={() => Alert.alert('Müzik', 'Şu anda dinlediğiniz müziği paylaşın!')}

            <Text style={styles.actionText}>Müzik</Text>

            style={styles.actionButton}

          <Text style={styles.infoTitle}>MSN İstatistikleri:</Text>
          <Text style={styles.infoText}>• 🟢 {demoContacts.filter(c => c.status.includes('🟢')).length} kişi çevrimiçi</Text>

  }

  if (currentScreen === 'chat') {
    return (
      <ScrollView style={styles.container}>
        {/* SOHBET BAŞLIK */}
        <View style={styles.chatHeader}>
          <TouchableOpacity onPress={() => setCurrentScreen('contacts')}>
            <Text style={styles.backButton}>← Geri</Text>
          </TouchableOpacity>

            <Text style={styles.chatUserName}>{selectedContact}</Text>
            <Text style={styles.chatUserStatus}>
              {demoContacts.find(c => c.name === selectedContact)?.status || '🟢 Çevrimiçi'}
            </Text>
          </View>
          <TouchableOpacity onPress={handleLogout}>
            <Text style={styles.closeButton}>✕</Text>
          </TouchableOpacity>
        </View>

        {/* MESAJLAR */}
        <View style={styles.messagesContainer}>
          {demoMessages.map(msg => (
            <View
              key={msg.id}
              style={[
                styles.messageBubble,
                msg.isMe ? styles.myMessage : styles.theirMessage

              {!msg.isMe && (

              <Text style={styles.messageText}>{msg.text}</Text>
              <Text style={styles.messageTime}>{msg.time}</Text>
            </View>
          ))}
        </View>

        {/* MESAJ GÖNDER */}
        <View style={styles.sendContainer}>
          <TouchableOpacity 
            style={styles.emojiMainButton}
            onPress={() => Alert.alert('Emoticons', 'Emoticon seçici yakında!')}
          >
            <Text style={styles.emojiButtonText}>😊</Text>
          </TouchableOpacity>
          <TextInput
            style={styles.messageInput}
            placeholder="Mesajınızı yazın..."
            value={messageText}
            onChangeText={setMessageText}
            placeholderTextColor="#999"
          />
          <TouchableOpacity
            style={styles.sendButton}
            onPress={() => {
              if (messageText.trim()) {
                Alert.alert('Başarılı', `${selectedContact}'a mesaj gönderildi: "${messageText}"`);
                setMessageText('');
              } else {
                Alert.alert('Hata', 'Lütfen mesaj yazın!');
              }
            }}
          >
            <Text style={styles.sendButtonText}>📨</Text>
          </TouchableOpacity>
        </View>

        {/* EMOJİLER */}
        <View style={styles.emojisContainer}>
          <Text style={styles.emojisTitle}>Hızlı Emoticons:</Text>
          <View style={styles.emojisRow}>
            {['😊', '😂', '😍', '😎', '🥳', '❤️', '🔥', '🎮'].map((emoji, i) => (
              <TouchableOpacity
                key={i}
                style={styles.emojiSmallButton}
                onPress={() => setMessageText(prev => prev + emoji)}
              >
                <Text style={styles.emojiText}>{emoji}</Text>
              </TouchableOpacity>
            ))}
          </View>
        </View>

        {/* EK ÖZELLİKLER */}
        <View style={styles.chatFeatures}>
          <TouchableOpacity 
            style={styles.chatFeature}
            onPress={() => Alert.alert('Dosya', 'Dosya gönderme özelliği yakında!')}
          >
            <Text style={styles.chatFeatureIcon}>📎</Text>
            <Text style={styles.chatFeatureText}>Dosya</Text>
          </TouchableOpacity>
          <TouchableOpacity 
            style={styles.chatFeature}
            onPress={() => Alert.alert('Resim', 'Resim gönderme özelliği yakında!')}
          >
            <Text style={styles.chatFeatureIcon}>📷</Text>
            <Text style={styles.chatFeatureText}>Resim</Text>
          </TouchableOpacity>
          <TouchableOpacity 
            style={styles.chatFeature}
            onPress={() => Alert.alert('Ses', 'Ses mesajı gönderme yakında!')}
          >
            <Text style={styles.chatFeatureIcon}>🎵</Text>
            <Text style={styles.chatFeatureText}>Ses</Text>
          </TouchableOpacity>
          <TouchableOpacity 
            style={styles.chatFeature}
            onPress={() => Alert.alert('Nudge!', `${selectedContact}'ı salladınız! 💥`)}
          >
            <Text style={styles.chatFeatureIcon}>👾</Text>
            <Text style={styles.chatFeatureText}>Nudge</Text>
          </TouchableOpacity>
        </View>

        {/* MSN HABERLERİ */}
        <View style={styles.newsBox}>
          <Text style={styles.newsTitle}>📰 MSN Haberleri:</Text>
          <Text style={styles.newsItem}>• 🎉 MSN Revival başarıyla çalışıyor!</Text>
          <Text style={styles.newsItem}>• 😊 Tüm butonlar aktif edildi</Text>
          <Text style={styles.newsItem}>• 🚀 Firebase entegrasyonu yakında</Text>
        </View>
      </ScrollView>
    );
  }

  return null;
}
